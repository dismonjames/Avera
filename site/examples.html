Ví dụ Mã nguồnCode Examples

1. FizzBuzz

fizzbuzz.av
#std.io

action main(): I32 will
    for i in 1..17 will
        if i % 15 == 0 will
            print("FizzBuzz")
        else if i % 3 == 0 will
            print("Fizz")
        else if i % 5 == 0 will
            print("Buzz")
        else
            print(i)
        finish
    finish
    return 0
finish

2. Magnet Pointer

magnet.av
#std.io

action main(): I32 will
    ::value = 42
    ~m = value
    print("Value:", value)
    print("Address:", ~m.address) // Lấy Address
    return 0
finish

3. Shape & Method

shape.av
#std.io

shape User will
    name: Text
    age:  I32
finish

action User.greet(&self): I32 will
    print("User:")
    print("name:", self.name)
    print("age:", self.age)
    return 0
finish

action make_user(n: Text, a: I32): User will
    :u = User(n, a)
    return u
finish

action main(): I32 will
    :me = make_user("Alice", 30)
    me.greet()
    return 0
finish
